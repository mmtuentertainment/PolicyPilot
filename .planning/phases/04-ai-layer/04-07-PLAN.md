    import { batchJobs } from '@/lib/db/schema';
    import { and, desc, eq, sql } from 'drizzle-orm';

    type BatchJobCreateInput = Omit<
      typeof batchJobs.$inferInsert,
      'orgId' | 'id' | 'createdAt' | 'updatedAt'
    >;

    type BatchJobStatus = 'in_progress' | 'completed' | 'failed';

      /**
       * Phase 4 D-06 — insert at Consistency-submit time.
       *
       * `status` defaults to 'in_progress' (Drizzle schema default per Plan 04-02). Caller may
       * pass it explicitly; if omitted, the default fires.

       * Anthropic batch IDs come from `client.messages.batches.create(...)` response and are
       * globally unique per Anthropic's namespace.
       */

        s.tx
          .insert(batchJobs)
          .values({
            ...input,
            orgId: s.orgId,
          })

       *   if (job.status === 'completed' || !isStale) return cached;

      findByAnthropicBatchId: async (s: OrgScope, anthropicBatchId: string) => {
        const rows = await s.tx
          .select()
          .from(batchJobs)
          .where(

              eq(batchJobs.orgId, s.orgId),
              eq(batchJobs.anthropicBatchId, anthropicBatchId),
            ),

              eq(batchJobs.type, 'consistency'),
