    Write a single script `scripts/check-foundation.ts` that runs all five success-criteria checks sequentially. The script must:

    1. **Print a per-criterion status line.** Format: `[1/5] OK  — tsc --noEmit zero errors` or `[1/5] FAIL — <reason>`.
    2. **Exit 0 only if all five pass.** Exit 1 if any fail.
    3. **Continue running all five checks even if one fails** — print all results so the operator sees the full failure set, not just the first. Track a `failed` boolean and exit 1 at the end if any check failed.

    ### Script structure

    ```
    // pnpm verify:phase-1 — runs all 5 ROADMAP Phase 1 success criteria.
    // Invoked with `--env-file=.env.local` so process.env is populated.
    import { spawnSync } from "node:child_process";
    import { sql } from "drizzle-orm";
    import { db } from "@/lib/db";

    const APP_URL = process.env.NEXT_PUBLIC_APP_URL ?? "http://localhost:3000";

    type Result = { ok: boolean; label: string; detail?: string };

    function logResult(idx: number, total: number, r: Result): void {
      const status = r.ok ? "OK  " : "FAIL";
      const detail = r.detail ? ` — ${r.detail}` : "";
      console.log(`[${idx}/${total}] ${status} — ${r.label}${detail}`);
    }

    async function checkTypecheck(): Promise<Result> {
      // Spawn `pnpm tsc --noEmit` synchronously.
      const result = spawnSync("pnpm", ["tsc", "--noEmit"], {
        encoding: "utf8",
        shell: true, // Windows compatibility — pnpm.cmd needs shell resolution.
      });
      if (result.status === 0) {
        return { ok: true, label: "tsc --noEmit zero errors" };
      }
      return {
        ok: false,
        label: "tsc --noEmit zero errors",
        detail: (result.stderr || result.stdout || "").trim().split("\n")[0] ?? "tsc failed",
      };
    }

    async function checkHttp(path: string, expectedStatus: number, label: string, bodyAssertion?: (body: string) => boolean): Promise<Result> {
      try {
        const res = await fetch(`${APP_URL}${path}`, { redirect: "manual" });
        if (res.status !== expectedStatus) {
          return {
            ok: false,
            label,
            detail: `expected ${expectedStatus}, got ${res.status}`,
          };
        }
        if (bodyAssertion) {
          const body = await res.text();
          if (!bodyAssertion(body)) {
            return {
              ok: false,
              label,
              detail: "body assertion failed (D-03 hero text missing)",
            };
          }
        }
        return { ok: true, label };
      } catch (err) {
        return {
          ok: false,
          label,
          detail: err instanceof Error ? err.message : "fetch failed (is `pnpm dev` running?)",
        };
      }
    }

    async function checkRedirect(path: string, expectedLocation: string, label: string): Promise<Result> {
      try {
        const res = await fetch(`${APP_URL}${path}`, { redirect: "manual" });
        if (res.status !== 307 && res.status !== 308 && res.status !== 302) {
          return {
            ok: false,
            label,
            detail: `expected 307 redirect, got ${res.status}`,
          };
        }
        const loc = res.headers.get("location") ?? "";
        if (!loc.includes(expectedLocation)) {
          return {
            ok: false,
            label,
            detail: `expected Location to include "${expectedLocation}", got "${loc}"`,
          };
        }
        return { ok: true, label };
      } catch (err) {
        return {
          ok: false,
          label,
          detail: err instanceof Error ? err.message : "fetch failed",
        };
      }
    }

    async function checkSelectOne(): Promise<Result> {
      try {
        const rows = await db.execute(sql`select 1 as ok`);
        const first = rows[0];
        if (!first || (first as { ok?: number }).ok !== 1) {
          return { ok: false, label: "Drizzle select 1 round-trip", detail: "unexpected result shape" };
        }
        return { ok: true, label: "Drizzle select 1 round-trip" };
      } catch (err) {
        return {
          ok: false,
          label: "Drizzle select 1 round-trip",
          detail: err instanceof Error ? err.message : "unknown DB error",
        };
      }
    }

    async function main(): Promise<void> {
      console.log("─── Phase 1 Foundation — verification ───");
      console.log(`App URL: ${APP_URL}`);
      console.log("");

      const results: Result[] = [];

      // Criterion 1: tsc --noEmit
      results.push(await checkTypecheck());
      logResult(1, 5, results[0]!);

      // Criterion 2: landing page loads with D-03 hero
      const c2 = await checkHttp(
        "/",
        200,
        "GET / returns 200 with D-03 hero copy",
        (body) => body.includes("Policy management for SMBs that beats a Google Drive folder"),
      );
      results.push(c2);
      logResult(2, 5, c2);

      // Criterion 3a: /sign-in renders
      const c3a = await checkHttp("/sign-in", 200, "GET /sign-in returns 200 (Clerk SignIn mount)");
      results.push(c3a);
      logResult(3, 5, c3a);

      // Criterion 3b: /sign-up renders (logged as supplemental to criterion 3)
      const c3b = await checkHttp("/sign-up", 200, "GET /sign-up returns 200 (Clerk SignUp mount)");
      results.push(c3b);
      logResult(4, 5, c3b);

      // Criterion 4: Drizzle select 1
      const c4 = await checkSelectOne();
      results.push(c4);
      logResult(5, 5, c4);

      // Criterion 5: middleware redirects private route to /sign-in
      // /sign-in-success is the canonical private placeholder (D-09); reaching
      // it unauthenticated must redirect.
      console.log("");
      console.log("─── Criterion 5 (extra): middleware redirect ───");
      const c5 = await checkRedirect("/sign-in-success", "/sign-in", "Middleware redirects /sign-in-success → /sign-in unauthenticated");
      results.push(c5);
      logResult(6, 6, c5);

      console.log("");
      const failed = results.filter((r) => !r.ok);
      if (failed.length > 0) {
        console.error(`✗ ${failed.length} of ${results.length} checks FAILED.`);
        for (const f of failed) {
          console.error(`  - ${f.label}${f.detail ? ` — ${f.detail}` : ""}`);
        }
        process.exit(1);
      }
      console.log(`✓ All ${results.length} checks passed. Phase 1 ready for Phase 2.`);
      process.exit(0);
    }

    void main();
    ```

    Note: the script lists 6 sub-checks but logs them across 5 ROADMAP criteria — criterion 3 covers both `/sign-in` and `/sign-up` (the script splits them for clarity in the failure output). Criterion 5 is logged separately as "extra" so the totals line is unambiguous; revise the indexing scheme if a different layout reads better — the contract is `process.exit(0)` only if every check passed.
