    Helper for the common "load + validate transition" sequence:
    ```typescript
    async function loadAndAssertTransition(
      s: { orgId: string; userId: string; role: PolicyStatus extends never ? never : 'admin' | 'reviewer' | 'employee'; tx: any },
      policyId: string,
      to: PolicyStatus,
    ): Promise<PolicyRow> {
      const rows = await Policies.findById(s as any, policyId);
      const policy = rows[0] as PolicyRow | undefined;
      if (!policy) throw new Error('Policy not found');
      if (!canTransition(policy.status, to)) {
        throw new IllegalTransitionError(policy.status, to);
      }
      return policy;
    }
    ```

    ```typescript
    export async function submitForReview(
      policyId: string,
      reviewerId: string | null,
    ): Promise<void> {
      const ctx = await getOrgContext();
      await withOrgScope(ctx, async (s) => {
        const policy = await loadAndAssertTransition(s, policyId, 'under_review');
        await WorkflowStages.recordSubmission(s, policy.id, reviewerId);
        await s.tx
          .update(policies)
          .set({ status: 'under_review', updatedAt: sql`now()` })
          .where(eq(policies.id, policyId));
      });
    }

    export async function approve(policyId: string): Promise<void> {
      // approve = under_review → published, same snapshot semantics as publish
      await publishCommon(policyId, 'under_review');
    }

    export async function reject(policyId: string, _reason?: string): Promise<void> {
      const ctx = await getOrgContext();
      await withOrgScope(ctx, async (s) => {
        await loadAndAssertTransition(s, policyId, 'draft');
        // (reject = under_review → draft; canTransition validates)
        await s.tx
          .update(policies)
          .set({ status: 'draft', updatedAt: sql`now()` })
          .where(eq(policies.id, policyId));
      });
    }

    export async function publish(policyId: string): Promise<void> {
      const ctx = await getOrgContext();
      await withOrgScope(ctx, async (s) => {
        const policy = await loadAndAssertTransition(s, policyId, 'published');
        // D-04: create policy_versions row capturing about-to-be-published content
        await PolicyVersions.create(s, {
          policyId: policy.id,
          versionNumber: policy.currentVersion,
          contentJson: policy.contentJson,
          createdBy: s.userId,
        });
        await s.tx
          .update(policies)
          .set({ status: 'published', updatedAt: sql`now()` })
          .where(eq(policies.id, policyId));
      });
    }

    async function publishCommon(policyId: string, _expectedFrom: PolicyStatus): Promise<void> {
      // Shared body for approve() = same as publish().
      // Kept separate to make Phase 6 tier-gating easy: approve will gate
      // on Growth+ approval workflow, publish will gate on Starter direct.
      await publish(policyId);
    }

    export async function archive(policyId: string): Promise<void> {
      const ctx = await getOrgContext();
      await withOrgScope(ctx, async (s) => {
        await loadAndAssertTransition(s, policyId, 'archived');
        await s.tx
          .update(policies)
          .set({ status: 'archived', updatedAt: sql`now()` })
          .where(eq(policies.id, policyId));
      });
    }

    export async function restore(policyId: string): Promise<void> {
      const ctx = await getOrgContext();
      await withOrgScope(ctx, async (s) => {
        await loadAndAssertTransition(s, policyId, 'draft');
        await s.tx
          .update(policies)
          .set({ status: 'draft', updatedAt: sql`now()` })
          .where(eq(policies.id, policyId));
      });
    }

    export async function editPublished(
      policyId: string,
      newContent: unknown,
      changeSummary?: string,
    ): Promise<void> {
      const ctx = await getOrgContext();
      await withOrgScope(ctx, async (s) => {
        const policy = await loadAndAssertTransition(s, policyId, 'draft');
        if (policy.status !== 'published') {
          // Belt-and-suspenders: loadAndAssertTransition already validated
          // via canTransition('published', 'draft') === true, but Plan 03-04's
          // ALLOWED_TRANSITIONS also allows other → 'draft' transitions
          // (under_review → draft via reject). editPublished is specifically
          // the published → draft path; reject the other entries here.
          throw new IllegalTransitionError(policy.status, 'draft');
        }
        // Snapshot the prior published version BEFORE overwriting (D-04 + L-05).
        await PolicyVersions.create(s, {
          policyId: policy.id,
          versionNumber: policy.currentVersion,
          contentJson: policy.contentJson,
          createdBy: s.userId,
          changeSummary,
        });
        await s.tx
          .update(policies)
          .set({
            contentJson: newContent,
            status: 'draft',
            currentVersion: policy.currentVersion + 1,
            updatedAt: sql`now()`,
          })
          .where(eq(policies.id, policyId));
      });
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Server Action → orchestrator | Server Action is the gateway; orchestrator is the authoritative state-machine gate |
| Orchestrator → withOrgScope tx | Single Postgres transaction enforces atomicity (snapshot+update happen together or not at all) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
