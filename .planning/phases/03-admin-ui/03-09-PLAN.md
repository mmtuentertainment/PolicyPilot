    ```typescript
    import { cookies, headers } from 'next/headers';
    import type { ReactNode } from 'react';
    import { OrganizationSwitcher, UserButton } from '@clerk/nextjs';
    import { SidebarProvider } from '@/components/ui/sidebar';
    import { AdminSidebar } from '@/components/admin/AdminSidebar';
    import { AdminTopbar } from '@/components/admin/AdminTopbar';
    import { requireAdmin } from '@/lib/auth/require-admin';

    /**
     * Admin route-group layout (D-06 shell + L-01 authoritative gate).
     *
     * requireAdmin() runs FIRST — calls notFound() (404) on non-admin role
     * per D-10 "advertise nothing". Defense-in-depth: middleware.ts also
     * gates via ADMIN_ROLE_REQUIRED_PATTERNS; this layout is the canonical
     * source.
     *
     * NOTE on /onboarding/create-org: that route lives at app/(admin)/onboarding/
     * but it MUST be reachable by signed-in users WITHOUT an admin role
     * (they're being onboarded). The route's own page.tsx (Plan 03-11) opts
     * out of the layout's requireAdmin() by either:
     *   (a) placing onboarding outside the (admin) group; OR
     *   (b) using a route-group escape hatch.
     * CONTEXT D-08 + UI-SPEC §Page-by-Page note this; the implementer
     * picks the structurally simpler option. Recommendation: place
     * /onboarding/create-org under (admin) but make this layout's
     * requireAdmin() skip when pathname === '/onboarding/create-org' via
     * the x-pathname header — avoids restructuring the file tree.
     */
    export default async function AdminLayout({ children }: { children: ReactNode }) {
      const headerStore = await headers();
      const pathname = headerStore.get('x-pathname') ?? '';

      // /onboarding/create-org bypass — user is mid-onboarding without an org.
      // Middleware still requires auth (ADMIN_URL_PATTERNS includes /onboarding)
      // but the layout-level admin-role gate would otherwise 404 a user who
      // hasn't yet been admin'd in their (about-to-be-created) org.
      const isOnboardingRoute = pathname.startsWith('/onboarding');
      if (!isOnboardingRoute) {
        await requireAdmin();
      }

      const cookieStore = await cookies();
      const defaultOpen = cookieStore.get('sidebar:state')?.value !== 'false';

      return (
        <SidebarProvider defaultOpen={defaultOpen}>
          <AdminSidebar />
          <main className="flex flex-col flex-1 min-h-screen">
            <AdminTopbar>
              <OrganizationSwitcher hidePersonal />
              <UserButton />
            </AdminTopbar>
            <div className="p-6 flex-1">{children}</div>
          </main>
        </SidebarProvider>
      );
    }
    ```

    Use the top-level static `import { cookies, headers } from 'next/headers'` (W9 closure). Both `cookies()` and `headers()` are async in Next.js 15 — await them inside the component body.

    The `hidePersonal` prop on `<OrganizationSwitcher />` is recommended by Clerk for B2B SaaS — hides the "Personal account" entry. If the prop name is different in our Clerk version (7.3.4), check Clerk docs and adjust. Falls back to no prop if uncertain.

    ```typescript
    import { headers } from 'next/headers';
    import Link from 'next/link';
    import type { ReactNode } from 'react';
    import { SidebarTrigger } from '@/components/ui/sidebar';

    /**
     * AdminTopbar — Server Component (D-06).
     *
     * Renders a 1-2-level breadcrumb derived from x-pathname (middleware
     * injection per Plan 03-02). The `children` slot accepts the layout-
     * passed Clerk widgets (<OrganizationSwitcher /> + <UserButton />).
     *
     * The SidebarTrigger gives the operator a Ctrl+B keyboard shortcut to
     * collapse/expand the sidebar.
     */
    function deriveBreadcrumbs(pathname: string): Array<{ label: string; href?: string }> {
      // Phase 3 ships static 1-2 level breadcrumbs; deeper trees are Phase 8 polish.
      if (pathname.startsWith('/dashboard')) return [{ label: 'Dashboard' }];
      if (pathname.startsWith('/policies/new')) {
        return [{ label: 'Policies', href: '/policies' }, { label: 'Create policy' }];
      }
      if (/^\/policies\/[^/]+$/.test(pathname)) {
        // /policies/[id] — second crumb is the policy's title, which we don't
        // have at the topbar level. Page-level breadcrumb override could
        // hydrate this in Phase 3.1; for now we use a generic label.
        return [{ label: 'Policies', href: '/policies' }, { label: 'Edit policy' }];
      }
      if (pathname.startsWith('/policies')) return [{ label: 'Policies' }];
      if (pathname.startsWith('/onboarding')) return [{ label: 'Onboarding' }];
      return [];
    }

    export async function AdminTopbar({ children }: { children: ReactNode }) {
      const headerStore = await headers();
      const pathname = headerStore.get('x-pathname') ?? '/';
      const crumbs = deriveBreadcrumbs(pathname);

      return (
        <header className="border-b">
          <div className="px-6 py-4 flex items-center justify-between gap-4">
            <div className="flex items-center gap-3 text-sm text-muted-foreground">
              <SidebarTrigger className="-ml-1" />
              {crumbs.map((c, i) => (
                <span key={`${c.label}-${i}`} className="flex items-center gap-3">
                  {c.href ? (
                    <Link href={c.href} className="hover:text-foreground">{c.label}</Link>
                  ) : (
                    <span className="text-foreground">{c.label}</span>
                  )}
                  {i < crumbs.length - 1 ? <span aria-hidden="true">/</span> : null}
                </span>
              ))}
            </div>
            <div className="flex items-center gap-2">{children}</div>
          </div>
        </header>
      );
    }
    ```

    The truncation logic for long breadcrumb labels (UI-SPEC mentions "max 50 chars then ellipsis") is deferred — Phase 3 ships static fallbacks for the [id] route. Operator can hydrate the policy title from the page-level via a `<title>`-passing context in Phase 8 if needed.
